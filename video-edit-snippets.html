MoviePy

The readable choice. Best when the edit is expressive — text overlays, fades, composites — and render time isn't the bottleneck.

clip-based · high-level
Load, trim & resize

OpenCV

Full frame-by-frame control. Best for computer-vision-driven edits — drawing, filtering, detection — where you need to touch every pixel.

frame-loop · low-level
Open source & writer 
import cv2
import numpy as np

cap = cv2.VideoCapture("input.mp4")
fps = cap.get(cv2.CAP_PROP_FPS)
width  = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))

fourcc = cv2.VideoWriter_fourcc(*'mp4v')
out = cv2.VideoWriter("output.mp4", fourcc, fps, (width, height))
Trim, draw & filter per frame 
frame_count = 0
start_frame = int(10 * fps)   # start at 10s
end_frame   = int(25 * fps)   # end at 25s

while True:
    ret, frame = cap.read()
    if not ret:
        break
    frame_count += 1

    if frame_count < start_frame or frame_count > end_frame:
        continue

    # frame = cv2.resize(frame, (1280, 720))

    cv2.putText(frame, "Hello World", (50, 100),
                cv2.FONT_HERSHEY_SIMPLEX, 2, (255, 255, 255), 3)

    cv2.rectangle(frame, (100, 100), (300, 300), (0, 255, 0), 2)

    # frame = cv2.GaussianBlur(frame, (15, 15), 0)
    # frame = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    # frame = cv2.cvtColor(frame, cv2.COLOR_GRAY2BGR)  # convert back for writer

    out.write(frame)

cap.release()
out.release()

ffmpeg-python

A thin wrapper over the ffmpeg binary. Best when speed matters and you don't need to touch pixels in Python at all.

binary wrapper · fastest
Trim & resize 
import ffmpeg

ffmpeg.input('input.mp4', ss=10, t=15).output('trimmed.mp4').run()

ffmpeg.input('input.mp4').filter('scale', 1280, 720).output('resized.mp4').run()
Mux audio & concatenate 
video = ffmpeg.input('video.mp4')
audio = ffmpeg.input('audio.mp3')
ffmpeg.output(video, audio, 'output.mp4', vcodec='copy', acodec='aac', shortest=None).run()

inputs = [ffmpeg.input(f) for f in ['part1.mp4', 'part2.mp4', 'part3.mp4']]
ffmpeg.concat(*inputs, v=1, a=1).output('merged.mp4').run()

Auto-Editor Script

Everything above, assembled: cut a list of segments, stitch them together, and drop in a title card.

moviepy · end-to-end
Full function 
from moviepy.editor import *
import os

def auto_editor(input_path, output_path, cuts):
    """
    cuts: list of (start, end) tuples in seconds
    """
    clip = VideoFileClip(input_path)
    segments = [clip.subclip(start, end) for start, end in cuts]

    final = concatenate_videoclips(segments, method="compose")

    txt = (TextClip("My Video", fontsize=100, color='white')
           .set_duration(3)
           .set_position('center')
           .fx(vfx.fadein, 1)
           .fx(vfx.fadeout, 1))

    final = concatenate_videoclips([txt, final])

    # music = AudioFileClip("bgm.mp3").subclip(0, final.duration).volumex(0.2)
    # final = final.set_audio(music)

    final.write_videofile(output_path, codec='libx264', threads=4)
    clip.close()

# Usage: keep only segments 5-10s and 20-30s
auto_editor("input.mp4", "output.mp4", [(5, 10), (20, 30)])
